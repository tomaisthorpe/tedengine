# Game States

Use the debug menu to push and pop game states.

```ts
const config = {
  states: { game: GameState, game2: DupeState },
  defaultState: 'game',
};

engine.gameState.push('game2');
engine.gameState.pop();
```

import { Suspense, use } from 'react';
import { browser } from 'react-dom';
import { TGame } from '@tedengine/ted';

export const Game = () => {
  use(browser());
  return (
    <TGame
      game={new Worker(new URL('@examples/game-states.ts', import.meta.url))}
    />
  );
};

<Suspense fallback={null}>
  <Game />
</Suspense>
