# Mouse

This example shows a box following the mouse location within the game window.

Mouse location and clicks are logged in console.

import { Suspense, use } from 'react';
import { browser } from 'react-dom';
import { TGame } from '@tedengine/ted';

export const Game = () => {
  use(browser());
  return (
    <TGame
      game={new Worker(new URL('@examples/input/mouse.ts', import.meta.url))}
    />
  );
};

<Suspense fallback={null}>
  <Game />
</Suspense>
