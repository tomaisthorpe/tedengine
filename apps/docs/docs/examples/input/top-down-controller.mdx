# Top Down Controller

This example shows how to build a custom system to calculate rotation of a sprite towards the mouse location.

import { Suspense, use } from 'react';
import { browser } from 'react-dom';
import { TGame } from '@tedengine/ted';

export const Game = () => {
  use(browser());
  return (
    <TGame
      game={
        new Worker(
          new URL('@examples/input/top-down-controller.ts', import.meta.url),
        )
      }
    />
  );
};

<Suspense fallback={null}>
  <Game />
</Suspense>
