# Debug Panels

This example shows how to use the various types of debug panels.

import { Suspense, use } from 'react';
import { browser } from 'react-dom';
import { TGame } from '@tedengine/ted';

export const Game = () => {
  use(browser());
  return (
    <TGame
      game={
        new Worker(new URL('@examples/debug/debug-panels.ts', import.meta.url))
      }
    />
  );
};

<Suspense fallback={null}>
  <Game />
</Suspense>
