# Sprite Ordering

Sprites are rendered in layers, the layer is a property of the sprite component.

In this example, pressing the `Flip` button will flip the sprite ordering.

import { Suspense, use } from 'react';
import { browser } from 'react-dom';
import { TGame } from '@tedengine/ted';

export const Game = () => {
  use(browser());
  return (
    <TGame
      game={
        new Worker(new URL('@examples/2d/sprite-ordering.ts', import.meta.url))
      }
    />
  );
};

<Suspense fallback={null}>
  <Game />
</Suspense>
