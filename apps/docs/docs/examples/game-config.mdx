# Game Config

Simple example to show what can be configured.
This example shows 100% container width with fixed 4:3 ratio and rendering at 400x300.
It also shows how to configure the colors for the loading screen and clear color.

```ts
const config = {
  states: { game: BoxState },
  defaultState: 'game',
  rendering: {
    clearColor: { r: 0.2, g: 0.1, b: 0.1, a: 1 },
  },
};

new TEngine(config, self as DedicatedWorkerGlobalScope);
```

import { Suspense, use } from 'react';
import { browser } from 'react-dom';
import { TGame } from '@tedengine/ted';

export const Game = () => {
  use(browser());
  return (
    <TGame
      width="100%"
      height="600px"
      config={{
        renderWidth: 400,
        renderHeight: 300,
        imageRendering: 'pixelated',
        loadingScreen: {
          backgroundColor: 'rgba(0, 0, 0, 0.5)',
          textColor: 'red',
        },
      }}
      game={new Worker(new URL('@examples/game-config.ts', import.meta.url))}
    />
  );
};

<Suspense fallback={null}>
  <Game />
</Suspense>
