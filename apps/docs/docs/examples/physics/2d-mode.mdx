# 2D Mode

Use WASD to move the box, it will be restricted to only move on the X/Y axis.

```ts
class GameState extends TGameState {
  public beforeWorldCreate() {
    this.world!.config.mode = '2d';
  }
}
```

import { Suspense, use } from 'react';
import { browser } from 'react-dom';
import { TGame } from '@tedengine/ted';

export const Game = () => {
  use(browser());
  return (
    <TGame
      game={
        new Worker(new URL('@examples/physics/2d-mode.ts', import.meta.url))
      }
    />
  );
};

<Suspense fallback={null}>
  <Game />
</Suspense>
